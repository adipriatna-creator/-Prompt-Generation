    <script>
        // Initialize Lucide Icons
        lucide.createIcons();

        // App State
        let currentFile = null;
        let apiKey = localStorage.getItem('GEMINI_API_KEY') || '';
        let generatedMasterPrompt = '';
        let generatedFullAnalysis = '';

        // UI Initialization
        document.addEventListener('DOMContentLoaded', () => {
            updateApiStatusUI();
            setupDragAndDrop();
        });

        // API Key Management
        function toggleApiModal(show) {
            const modal = document.getElementById('api-modal');
            if (show) {
                document.getElementById('api-key-input').value = apiKey;
                modal.classList.remove('hidden');
            } else {
                modal.classList.add('hidden');
            }
        }

        function saveApiKey() {
            const inputKey = document.getElementById('api-key-input').value.trim();
            if (!inputKey) {
                showToast('Silakan masukkan API Key Gemini yang valid.', 'error');
                return;
            }
            apiKey = inputKey;
            localStorage.setItem('GEMINI_API_KEY', apiKey);
            updateApiStatusUI();
            toggleApiModal(false);
            showToast('API Key berhasil disimpan!', 'success');
        }

        function updateApiStatusUI() {
            const statusText = document.getElementById('api-status-text');
            if (apiKey) {
                statusText.innerText = 'API Key Terpasang';
                statusText.classList.add('text-emerald-400');
            } else {
                statusText.innerText = 'Set Gemini API Key';
                statusText.classList.remove('text-emerald-400');
            }
        }

        // Drag and Drop Handling
        function setupDragAndDrop() {
            const zone = document.getElementById('upload-zone');
            
            ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(eventName => {
                zone.addEventListener(eventName, preventDefaults, false);
            });

            function preventDefaults(e) {
                e.preventDefault();
                e.stopPropagation();
            }

            ['dragenter', 'dragover'].forEach(eventName => {
                zone.addEventListener(eventName, () => zone.classList.add('border-brand-400', 'bg-slate-900/70'), false);
            });

            ['dragleave', 'drop'].forEach(eventName => {
                zone.addEventListener(eventName, () => zone.classList.remove('border-brand-400', 'bg-slate-900/70'), false);
            });

            zone.addEventListener('drop', (e) => {
                const dt = e.dataTransfer;
                const files = dt.files;
                if (files && files.length > 0) {
                    processFile(files[0]);
                }
            });

            zone.addEventListener('click', () => {
                document.getElementById('video-input').click();
            });
        }

        function handleFileSelect(e) {
            const files = e.target.files;
            if (files && files.length > 0) {
                processFile(files[0]);
            }
        }

        function processFile(file) {
            if (!file.type.startsWith('video/')) {
                showToast('Format video tidak didukung. Harap unggah file video.', 'error');
                return;
            }

            currentFile = file;

            // Render Preview
            const videoUrl = URL.createObjectURL(file);
            const videoEl = document.getElementById('video-preview');
            videoEl.src = videoUrl;

            // Metadata updates
            document.getElementById('meta-name').innerText = file.name;
            document.getElementById('meta-size').innerText = formatBytes(file.size);

            videoEl.onloadedmetadata = () => {
                const duration = formatDuration(videoEl.duration);
                document.getElementById('meta-duration').innerText = duration;
                document.getElementById('meta-resolution').innerText = `${videoEl.videoWidth} x ${videoEl.videoHeight}`;
            };

            // Switch Views
            document.getElementById('upload-zone').classList.add('hidden');
            document.getElementById('preview-container').classList.remove('hidden');
            document.getElementById('result-section').classList.add('hidden');
            showToast('Video berhasil diunggah!', 'success');
        }

        function removeVideo() {
            currentFile = null;
            const videoEl = document.getElementById('video-preview');
            videoEl.pause();
            videoEl.src = '';
            
            document.getElementById('upload-zone').classList.remove('hidden');

        // Helpers
        function formatBytes(bytes) {
            if (bytes === 0) return '0 Bytes';
            const k = 1024;
            const sizes = ['Bytes', 'KB', 'MB', 'GB'];
            const i = Math.floor(Math.log(bytes) / Math.log(k));
            return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
        }

        function formatDuration(seconds) {
            const mins = Math.floor(seconds / 60);
            const secs = Math.floor(seconds % 60);
            return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
        }

        // Base64 Converter for File API fallback / Gemini inlineData
        function fileToBase64(file) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.readAsDataURL(file);
                reader.onload = () => {
                    const base64String = reader.result.split(',')[1];
                    resolve(base64String);
                };
                reader.onerror = error => reject(error);
            });
        }

        // Parse Output text into Master Prompt & Detailed Categories
        function parseAndDisplayResult(fullText) {
            // Sanitize text to remove any remaining double-asterisk Markdown bolding
            const cleanText = fullText.replace(/\*\*/g, '');

            let masterPrompt = "";
            let fullAnalysis = cleanText;

            if (cleanText.includes("=== MASTER PROMPT SIAP GENERATE ===")) {
                const parts = cleanText.split("=== MASTER PROMPT SIAP GENERATE ===");
                fullAnalysis = parts[0].replace("=== RINCIAN ANALISIS DETAIL ===", "").trim();
                masterPrompt = parts[1].trim();
            } else {
                masterPrompt = cleanText;
            }

            generatedMasterPrompt = masterPrompt;
            generatedFullAnalysis = fullAnalysis;

            document.getElementById('master-prompt-text').value = masterPrompt;
            document.getElementById('full-analysis-text').value = fullAnalysis;

            document.getElementById('result-section').classList.remove('hidden');
            document.getElementById('result-section').scrollIntoView({ behavior: 'smooth' });
        }

        function copyToClipboard(text) {
            // Using document.execCommand('copy') directly as navigator.clipboard.writeText is restricted in iframe preview environments
            const textArea = document.createElement("textarea");
            textArea.value = text;
            // Prevent scrolling to bottom
            textArea.style.top = "0";
            textArea.style.left = "0";
            textArea.style.position = "fixed";
            textArea.style.opacity = "0";
            document.body.appendChild(textArea);
            textArea.focus();
            textArea.select();
            try {
                const successful = document.execCommand('copy');
                if (!successful) {
                    console.warn('execCommand copy fallback was unsuccessful');
                }
            } catch (err) {
                console.error('Copy to clipboard failed:', err);
            }
            document.body.removeChild(textArea);
        }
